name: Paleramine Runtime Preview OTA Installer

on:
  workflow_dispatch:
  push:
    branches:
      - automation/apple-signing-kit
    paths:
      - ".github/workflows/dopamine-profile-ota.yml"
      - "tool/prepare_dopamine_profile_ota.py"
      - "tool/dopamine_one_tap_server.py"

permissions:
  contents: write
  actions: read

concurrency:
  group: paleramine-profile-ota-live
  cancel-in-progress: true

jobs:
  installer:
    name: Build safe Paleramine Runtime Preview and serve signed IPA
    runs-on: macos-latest
    timeout-minutes: 330
    env:
      APP_STORE_CONNECT_API_KEY_P8: ${{ secrets.APP_STORE_CONNECT_API_KEY_P8 }}
      APP_STORE_CONNECT_KEY_ID: ${{ secrets.APP_STORE_CONNECT_KEY_ID }}
      APP_STORE_CONNECT_ISSUER_ID: ${{ secrets.APP_STORE_CONNECT_ISSUER_ID }}
      APPLE_P12_PASSWORD: "1"
      PALERAMINE_BUNDLE_ID: com.nightvibes33.paleramine.ipad5
      PALERAMINE_PROFILE_NAME: Paleramine iPad 5 Development
      FALLBACK_SIGNING_ARTIFACT_ID: "8452841516"
      SOURCE_REPOSITORY: NightVibes33/Dopamine
      SOURCE_BRANCH: research/paleramine-runtime-preentry

    steps:
      - name: Checkout installer
        uses: actions/checkout@v4

      - name: Validate existing Apple credentials
        shell: bash
        run: |
          set -euo pipefail
          missing=0
          for name in APP_STORE_CONNECT_API_KEY_P8 APP_STORE_CONNECT_KEY_ID APP_STORE_CONNECT_ISSUER_ID; do
            if [[ -z "${!name:-}" ]]; then
              echo "::error::Missing existing mo1 secret: $name"
              missing=1
            fi
          done
          exit "$missing"

      - name: Select current Xcode
        uses: maxim-lobanov/setup-xcode@v1
        with:
          xcode-version: latest-stable

      - name: Install Procursus build tools
        uses: dhinakg/procursus-action@main
        with:
          packages: ldid findutils sed coreutils make

      - name: Install build, registration, and tunnel dependencies
        shell: bash
        run: |
          set -euxo pipefail
          brew install make libarchive openssl cloudflared
          echo "PATH=$(brew --prefix make)/libexec/gnubin:$PATH" >> "$GITHUB_ENV"
          python3 -m venv .venv
          .venv/bin/python -m pip install --disable-pip-version-check --quiet pyjwt cryptography requests

      - name: Clone pinned Paleramine source branch
        shell: bash
        run: |
          set -euxo pipefail
          EXPECTED_SHA="$(git ls-remote "https://github.com/${SOURCE_REPOSITORY}.git" "refs/heads/${SOURCE_BRANCH}" | awk '{print $1}')"
          test -n "$EXPECTED_SHA"
          git clone --depth 1 --branch "$SOURCE_BRANCH" --recurse-submodules --shallow-submodules \
            "https://github.com/${SOURCE_REPOSITORY}.git" paleramine
          ACTUAL_SHA="$(git -C paleramine rev-parse HEAD)"
          [[ "$ACTUAL_SHA" == "$EXPECTED_SHA" ]]
          echo "PALERAMINE_SOURCE_SHA=$ACTUAL_SHA" >> "$GITHUB_ENV"
          printf '%s\n' "$ACTUAL_SHA" > paleramine-installer-source-sha.txt

      - name: Apply Paleramine branding and safe runtime-preview gate
        shell: bash
        working-directory: paleramine
        run: |
          set -euxo pipefail
          python3 -m py_compile research/paleramine_full_build_patch.py
          python3 research/paleramine_full_build_patch.py
          python3 -m py_compile research/paleramine_runtime_preview_patch.py
          python3 research/paleramine_runtime_preview_patch.py
          git diff --check
          test -s paleramine-build-manifest.txt
          grep -F 'brand=Paleramine' paleramine-build-manifest.txt
          grep -F 'bundle_id=com.nightvibes33.paleramine.ipad5' paleramine-build-manifest.txt
          grep -F 'build_mode=runtime-preview' paleramine-build-manifest.txt
          grep -F 'kernel_entry=disabled' paleramine-build-manifest.txt
          grep -F 'ota_installable=true' paleramine-build-manifest.txt

      - name: Install THEOS and iOS 16.5 SDK
        shell: bash
        working-directory: paleramine
        run: |
          set -euxo pipefail
          export THEOS="$PWD/theos"
          mkdir -p "$THEOS"
          curl -fsSL https://raw.githubusercontent.com/theos/theos/master/bin/install-theos > install-theos.sh
          gsed -E "/^\s*get_theos\s*$/,+1 s/^(\s*)(get_sdks)\s*$/\1mkdir -p \${THEOS}\/sdks\n\1touch \${THEOS}\/sdks\/sdk\n\1\2/g" -i install-theos.sh
          bash install-theos.sh
          curl -fL https://github.com/theos/sdks/releases/latest/download/iPhoneOS16.5.sdk.tar.xz \
            -o "$THEOS/sdks/iPhoneOS16.5.sdk.tar.xz"
          tar -xf "$THEOS/sdks/iPhoneOS16.5.sdk.tar.xz" -C "$THEOS/sdks"
          rm "$THEOS/sdks/iPhoneOS16.5.sdk.tar.xz"

      - name: Build and install trustcache
        shell: bash
        working-directory: paleramine
        run: |
          set -euxo pipefail
          git clone --depth 1 https://github.com/CRKatri/trustcache
          cd trustcache
          export CFLAGS="${CFLAGS:-} -I$(brew --prefix openssl)/include -arch arm64"
          export LDFLAGS="${LDFLAGS:-} -L$(brew --prefix openssl)/lib -arch arm64"
          gmake -j"$(sysctl -n hw.logicalcpu)" OPENSSL=1
          sudo cp trustcache /opt/procursus/bin/

      - name: Set Paleramine build metadata
        shell: bash
        working-directory: paleramine
        run: |
          set -euxo pipefail
          T2="$(TZ=UTC-2 date +'%Y%m%d_%H%M%S')"
          TS="$(date -j -f '%Y%m%d_%H%M%S' "$T2" +%s)"
          SHASH="$(git rev-parse --short HEAD)"
          echo "ctime=$T2" >> "$GITHUB_ENV"
          echo "ctimestamp=$TS" >> "$GITHUB_ENV"
          echo "shorthash=$SHASH" >> "$GITHUB_ENV"

      - name: Download bootstrap assets
        shell: bash
        working-directory: paleramine/Application/Dopamine/Resources
        run: |
          set -euxo pipefail
          ./download_bootstraps.sh

      - name: Build Paleramine IPA and symbols
        shell: bash
        working-directory: paleramine
        run: |
          set -euxo pipefail
          export BASEDIR="$PWD"
          export THEOS="$BASEDIR/theos"
          gmake -j"$(sysctl -n hw.logicalcpu)" NIGHTLY=1
          test -s Application/Paleramine.ipa
          mkdir -p "$GITHUB_WORKSPACE/installer-output" "$GITHUB_WORKSPACE/signing"
          shasum -a 256 Application/Paleramine.ipa | tee "$GITHUB_WORKSPACE/installer-output/Paleramine-unsigned.sha256"

          APP_BIN="Application/build/Build/Products/Debug-iphoneos/Paleramine.app/Paleramine"
          DS_BIN="Application/build/Build/Products/Debug-iphoneos/Paleramine.app/Frameworks/DarkSword.framework/DarkSword"
          test -s "$APP_BIN"
          test -s "$DS_BIN"
          dsymutil "$APP_BIN" -o "$GITHUB_WORKSPACE/installer-output/Paleramine.app.dSYM"
          dsymutil "$DS_BIN" -o "$GITHUB_WORKSPACE/installer-output/DarkSword.framework.dSYM"
          dwarfdump --uuid "$APP_BIN" | tee "$GITHUB_WORKSPACE/installer-output/Paleramine-binary-uuid.txt"
          dwarfdump --uuid "$DS_BIN" | tee "$GITHUB_WORKSPACE/installer-output/DarkSword-binary-uuid.txt"
          cp paleramine-build-manifest.txt "$GITHUB_WORKSPACE/installer-output/"

      - name: Restore existing Apple development P12
        env:
          GH_TOKEN: ${{ github.token }}
        shell: bash
        run: |
          set -euxo pipefail
          SIGNING_ARTIFACT_ID="$(gh api "/repos/${GITHUB_REPOSITORY}/actions/artifacts?name=Dopamine-fast-session&per_page=100" \
            --jq '[.artifacts[] | select(.expired == false)] | sort_by(.created_at) | reverse | .[0].id // empty' 2>/dev/null || true)"
          SIGNING_ARTIFACT_ID="${SIGNING_ARTIFACT_ID:-$FALLBACK_SIGNING_ARTIFACT_ID}"
          gh api "/repos/${GITHUB_REPOSITORY}/actions/artifacts/${SIGNING_ARTIFACT_ID}/zip" > /tmp/signing-artifact.zip
          rm -rf /tmp/signing-artifact
          mkdir -p /tmp/signing-artifact signing
          unzip -q /tmp/signing-artifact.zip -d /tmp/signing-artifact
          P12_SOURCE="$(find /tmp/signing-artifact -name 'Dopamine-Development.p12' -type f | head -n 1)"
          test -n "$P12_SOURCE"
          cp "$P12_SOURCE" signing/Paleramine-Development.p12
          openssl pkcs12 -legacy \
            -in signing/Paleramine-Development.p12 \
            -clcerts -nokeys \
            -passin pass:"$APPLE_P12_PASSWORD" \
            -out signing/development.pem
          P12_CERT_SERIAL="$(openssl x509 -in signing/development.pem -serial -noout | cut -d= -f2)"
          export P12_CERT_SERIAL

          .venv/bin/python - <<'PY'
          import os
          import re
          import time
          from pathlib import Path
          import jwt
          import requests

          def normalize(value: str) -> str:
              return re.sub(r"[^0-9A-Fa-f]", "", str(value or "")).upper().lstrip("0") or "0"

          key = os.environ["APP_STORE_CONNECT_API_KEY_P8"].replace("\\n", "\n")
          now = int(time.time())
          token = jwt.encode(
              {"iss": os.environ["APP_STORE_CONNECT_ISSUER_ID"], "iat": now, "exp": now + 900, "aud": "appstoreconnect-v1"},
              key,
              algorithm="ES256",
              headers={"kid": os.environ["APP_STORE_CONNECT_KEY_ID"], "typ": "JWT"},
          )
          headers = {"Authorization": f"Bearer {token}"}
          target = normalize(os.environ["P12_CERT_SERIAL"])
          url = "https://api.appstoreconnect.apple.com/v1/certificates"
          params = {"limit": 200}
          matches = []
          development = []
          while url:
              response = requests.get(url, headers=headers, params=params, timeout=60)
              response.raise_for_status()
              payload = response.json()
              for item in payload.get("data", []):
                  attrs = item.get("attributes", {})
                  cert_type = str(attrs.get("certificateType") or "")
                  if cert_type in {"IOS_DEVELOPMENT", "DEVELOPMENT"}:
                      development.append(item)
                  if normalize(attrs.get("serialNumber")) == target:
                      matches.append(item)
              url = payload.get("links", {}).get("next")
              params = None
          if matches:
              selected = matches[0]
          elif len(development) == 1:
              selected = development[0]
          else:
              raise SystemExit(f"Could not match preserved P12 serial {target} to an Apple development certificate")
          Path("signing/certificate.id").write_text(selected["id"], encoding="utf-8")
          PY

          echo "APPLE_CERTIFICATE_ID=$(cat signing/certificate.id)" >> "$GITHUB_ENV"
          echo "APPLE_P12_PATH=$PWD/signing/Paleramine-Development.p12" >> "$GITHUB_ENV"

      - name: Import signing identity
        shell: bash
        run: |
          set -euxo pipefail
          KEYCHAIN="$RUNNER_TEMP/paleramine-profile-ota.keychain-db"
          KEYCHAIN_PASSWORD="$(openssl rand -hex 24)"
          security create-keychain -p "$KEYCHAIN_PASSWORD" "$KEYCHAIN"
          security set-keychain-settings -lut 21600 "$KEYCHAIN"
          security unlock-keychain -p "$KEYCHAIN_PASSWORD" "$KEYCHAIN"
          security import "$APPLE_P12_PATH" -k "$KEYCHAIN" -P "$APPLE_P12_PASSWORD" -A -T /usr/bin/codesign -T /usr/bin/security
          security set-key-partition-list -S apple-tool:,apple:,codesign: -s -k "$KEYCHAIN_PASSWORD" "$KEYCHAIN"
          security list-keychains -d user -s "$KEYCHAIN" $(security list-keychains -d user | tr -d '"')
          IDENTITY="$(security find-identity -v -p codesigning "$KEYCHAIN" | grep -Eo '[0-9A-F]{40}' | head -n 1)"
          test -n "$IDENTITY"
          echo "APPLE_SIGNING_IDENTITY=$IDENTITY" >> "$GITHUB_ENV"
          echo "SIGNING_KEYCHAIN=$KEYCHAIN" >> "$GITHUB_ENV"

      - name: Start Paleramine registration and OTA server
        shell: bash
        run: |
          set -euxo pipefail
          .venv/bin/python tool/prepare_dopamine_profile_ota.py
          .venv/bin/python -m py_compile tool/dopamine_one_tap_server.py
          SESSION="$(openssl rand -hex 20)"
          CHALLENGE="$(openssl rand -hex 32)"
          export INSTALLER_SESSION="$SESSION"
          export INSTALLER_CHALLENGE="$CHALLENGE"
          export UNSIGNED_IPA="$PWD/paleramine/Application/Paleramine.ipa"
          export OUTPUT_DIR="$PWD/installer-output"
          export PALERAMINE_BUILD_NUMBER="$(date -u +%Y%m%d%H%M%S)"
          nohup .venv/bin/python tool/dopamine_one_tap_server.py > installer-output/server.log 2>&1 &
          echo $! > installer-output/server.pid
          for _ in $(seq 1 30); do
            curl -fsS http://127.0.0.1:8080/health >/dev/null && break
            sleep 1
          done
          curl -fsS http://127.0.0.1:8080/health

      - name: Publish Paleramine Cloudflare installer
        env:
          GH_TOKEN: ${{ github.token }}
        shell: bash
        run: |
          set -euxo pipefail
          nohup cloudflared tunnel --url http://127.0.0.1:8080 --no-autoupdate > installer-output/tunnel.log 2>&1 &
          echo $! > installer-output/tunnel.pid
          URL=""
          for _ in $(seq 1 90); do
            URL="$(grep -Eo 'https://[-a-z0-9]+\.trycloudflare\.com' installer-output/tunnel.log | head -n 1 || true)"
            [[ -n "$URL" ]] && break
            sleep 1
          done
          test -n "$URL"
          printf '%s/\n' "$URL" > paleramine-installer-url.txt
          printf '%s\n' "$PALERAMINE_SOURCE_SHA" > paleramine-installer-source-sha.txt
          echo "INSTALLER_URL=$URL/" >> "$GITHUB_ENV"
          echo "::notice title=Paleramine Profile OTA Installer::$URL/"
          echo "::notice title=Paleramine Source Commit::$PALERAMINE_SOURCE_SHA"

          upload_file() {
            local path="$1"
            local message="$2"
            local encoded current_sha
            encoded="$(base64 < "$path" | tr -d '\n')"
            current_sha="$(gh api "repos/${GITHUB_REPOSITORY}/contents/${path}?ref=automation/apple-signing-kit" --jq .sha 2>/dev/null || true)"
            if [[ -n "$current_sha" ]]; then
              gh api --method PUT "repos/${GITHUB_REPOSITORY}/contents/${path}" \
                -f message="$message" -f content="$encoded" \
                -f branch='automation/apple-signing-kit' -f sha="$current_sha" >/dev/null
            else
              gh api --method PUT "repos/${GITHUB_REPOSITORY}/contents/${path}" \
                -f message="$message" -f content="$encoded" \
                -f branch='automation/apple-signing-kit' >/dev/null
            fi
          }
          upload_file paleramine-installer-url.txt 'Update Paleramine profile OTA URL'
          upload_file paleramine-installer-source-sha.txt 'Record Paleramine source SHA'

      - name: Upload installer URL and source proof
        uses: actions/upload-artifact@v4
        with:
          name: Paleramine-profile-OTA-installer-url
          path: |
            paleramine-installer-url.txt
            paleramine-installer-source-sha.txt
          retention-days: 1
          if-no-files-found: error

      - name: Keep installer online
        shell: bash
        run: |
          set -euo pipefail
          echo "PALERAMINE_INSTALLER_URL=$INSTALLER_URL"
          echo "PALERAMINE_SOURCE_SHA=$PALERAMINE_SOURCE_SHA"
          SERVER_PID="$(cat installer-output/server.pid)"
          TUNNEL_PID="$(cat installer-output/tunnel.pid)"
          END=$((SECONDS + 14400))
          while [[ "$SECONDS" -lt "$END" ]]; do
            kill -0 "$SERVER_PID" 2>/dev/null || { cat installer-output/server.log; exit 1; }
            kill -0 "$TUNNEL_PID" 2>/dev/null || { cat installer-output/tunnel.log; exit 1; }
            sleep 30
          done

      - name: Upload Paleramine diagnostics
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: Paleramine-profile-OTA-session
          path: |
            installer-output/
            signing/Paleramine-Development.p12
            paleramine-installer-source-sha.txt
            paleramine/paleramine-build-manifest.txt
          retention-days: 1
          if-no-files-found: warn
